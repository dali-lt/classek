// ===== À AJOUTER dans firestore.rules, à l'intérieur de "match /databases/{database}/documents { ... }" =====
// (à côté des autres "match /users/..." ; garde ton uid admin dans isAdmin() comme avant)

// Boîte à idées + avis par exercice (élève -> page admin)
match /feedback/{id} {
  allow read: if isAdmin() || (request.auth != null && resource.data.uid == request.auth.uid);

  allow create: if request.auth != null
    && request.resource.data.uid == request.auth.uid
    && request.resource.data.keys().hasOnly(['kind','uid','role','name','level','type','message','chapterId','exerciseId','rating','status','createdAt'])
    && request.resource.data.kind in ['message','rating']
    && request.resource.data.role in ['student','parent','teacher']
    && request.resource.data.status == 'new'
    && (request.resource.data.kind != 'message' || (
          request.resource.data.type in ['difficulty','clarity','motivation','idea','bug','other']
          && request.resource.data.message is string
          && request.resource.data.message.size() >= 3
          && request.resource.data.message.size() <= 1000))
    && (request.resource.data.kind != 'rating' || (
          request.resource.data.rating in ['easy','ok','hard']
          && request.resource.data.chapterId is string
          && request.resource.data.exerciseId is string
          && id == 'r_' + request.auth.uid + '_' + request.resource.data.chapterId + '_' + request.resource.data.exerciseId));

  // l'élève peut changer SON avis (même exercice); l'admin peut tout modifier (statut, note)
  allow update: if isAdmin() || (
    request.auth != null
    && resource.data.uid == request.auth.uid
    && resource.data.kind == 'rating'
    && request.resource.data.uid == request.auth.uid
    && request.resource.data.kind == 'rating'
    && request.resource.data.rating in ['easy','ok','hard']
    && request.resource.data.status == 'new'
    && request.resource.data.chapterId == resource.data.chapterId
    && request.resource.data.exerciseId == resource.data.exerciseId
    && request.resource.data.keys().hasOnly(['kind','uid','role','name','level','rating','chapterId','exerciseId','status','createdAt']));

  allow delete: if isAdmin();
}
